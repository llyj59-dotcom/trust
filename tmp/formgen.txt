/* ---------------- 공통 ---------------- */
function 반칸(form) {
  return form.addTextItem()
    .setTitle('반')
    .setHelpText('우리 반을 적어 주세요. (예: 6-8)')
    .setRequired(true);
}

function 참여코드칸(form, 도움말) {
  var item = form.addTextItem().setTitle('참여 코드').setHelpText(도움말).setRequired(true);
  item.setValidation(
    FormApp.createTextValidation()
      .setHelpText('K와 숫자 두 자리로 써 주세요. (예: K07)')
      .requireTextMatchesPattern('^[Kk][0-9]{2}$')
      .build()
  );
  return item;
}

function 시트로보내기(form, 시트ID, 탭이름) {
  try {
    form.setDestination(FormApp.DestinationType.SPREADSHEET, 시트ID);
    SpreadsheetApp.flush();
    var ss = SpreadsheetApp.openById(시트ID);
    var sheets = ss.getSheets();
    var last = sheets[sheets.length - 1];
    var n = last.getName();
    if (n.indexOf('설문지 응답') === 0 || n.indexOf('Form Responses') === 0) last.setName(탭이름);
  } catch (e) {
    Logger.log('응답 시트 연결은 건너뛰었어요: ' + e.message);
  }
}

/* 필요할 때만 — 로그인 없이 답할 수 있게 풀기 (조직 정책으로 막혀 있을 수 있습니다) */
function 제한풀기() {
  var ids = [];   // 폼 '고치는 주소'의 /d/ 와 /edit 사이 값을 여기에 넣으세요
  var out = [];
  ids.forEach(function (id) {
    var f = FormApp.openById(id);
    var 전 = f.requiresLogin();
    try { f.setRequireLogin(false); }
    catch (e) { out.push(f.getTitle() + ' : 풀 수 없음 — ' + e.message); return; }
    out.push(f.getTitle() + ' : 로그인 필요 ' + 전 + ' → ' + f.requiresLogin());
  });
  Logger.log(out.join('\n') || 'ids 배열에 폼 ID를 넣어 주세요.');
}
